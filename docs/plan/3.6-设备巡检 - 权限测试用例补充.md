#### **TC-PERM-002: 维保人员可提交巡检记录但不可创建计划**

```python
def test_maintainer_can_submit_record_but_not_create_plan(
    client: TestClient, db: Session, get_maintainer_token: str,
    inspection_task_fixture: InspectionTask
):
    """维保人员有 inspection:execute 权限，可提交记录但无 inspection:create"""
    
    # ✅ 提交记录成功
    response = client.post(
        f"/api/v1/inspection-tasks/{inspection_task_fixture.id}/records",
        json={
            "device_id": 12,
            "result": "normal"
        },
        headers={"Authorization": f"Bearer {get_maintainer_token}"}
    )
    assert response.status_code == 200
    
    # ❌ 创建计划被拒绝
    response = client.post(
        "/api/v1/inspection-plans",
        json={
            "plan_name": "测试计划",
            "cycle_type": "daily",
            "responsible_user_id": 1
        },
        headers={"Authorization": f"Bearer {get_maintainer_token}"}
    )
    assert response.status_code == 403
```

**覆盖点**: FR-032/FR-034 权限分离

#### **TC-PERM-003: 消防主管可访问所有巡检接口**

```python
def test_chief_has_full_access_to_inspection_modules(
    client: TestClient, db: Session, get_superuser_token: str
):
    """消防主管拥有 inspection:view/create/update/delete/execute 全部权限"""
    
    # ✅ 创建计划
    response = client.post(
        "/api/v1/inspection-plans",
        json={"plan_name": "测试计划", "cycle_type": "daily", "responsible_user_id": 1},
        headers={"Authorization": f"Bearer {get_superuser_token}"}
    )
    assert response.status_code == 200
    
    # ✅ 更新计划
    plan_id = response.json()["data"]["id"]
    response = client.put(
        f"/api/v1/inspection-plans/{plan_id}",
        json={"plan_name": "更新后的计划"},
        headers={"Authorization": f"Bearer {get_superuser_token}"}
    )
    assert response.status_code == 200
    
    # ✅ 删除计划
    response = client.delete(
        f"/api/v1/inspection-plans/{plan_id}",
        headers={"Authorization": f"Bearer {get_superuser_token}"}
    )
    assert response.status_code == 200
    
    # ✅ 查看所有接口
    for endpoint in [
        "/api/v1/inspection-plans",
        "/api/v1/inspection-tasks",
        "/api/v1/inspection-records",
        "/api/v1/inspection-missed-stats"
    ]:
        response = client.get(endpoint, headers={"Authorization": f"Bearer {get_superuser_token}"})
        assert response.status_code == 200
```

**覆盖点**: FR-032~FR-035 全模块权限

#### **TC-PERM-004: 数据范围过滤 - 仅看本部门及子部门设备**

```python
@pytest.mark.parametrize("role,org_scope", [
    ("chief", "all"),  # 主管可查看所有区域
    ("maintainer", "dept"),  # 维保人员只看本部门
])
def test_data_scope_filtering_on_tasks(
    client: TestClient, db: Session, role: str, org_scope: str,
    get_token_by_role: dict[str, str], inspection_plan_with_multiple_orgs
):
    """巡检任务按用户 data_scope 过滤，superuser=all, maintainer=dept"""
    token = get_token_by_role[role]
    response = client.get(
        "/api/v1/inspection-tasks",
        params={"page": 1, "page_size": 20},
        headers={"Authorization": f"Bearer {token}"}
    )
    
    assert response.status_code == 200
    tasks = response.json()["data"]["items"]
    
    if org_scope == "all":
        # 主管应看到所有区域的任務
        assert len(tasks) > 0
    else:
        # 维保人员只能看到自己部门的任務
        assert all(task["org_id"] in user_visible_orgs for task in tasks)
```

**覆盖点**: DEC-004 数据权限规范

def test_inspection_records_created_by_filter(

## 三、前端权限测试补充

### 3.1 Vue 组件权限指令测试

```javascript
// frontend/src/views/inspection/__tests__/Plan.spec.js
import { mount } from '@vue/test-utils'
import PermissionDirective from '@/directives/permission'
import Plan.vue from '../Plan.vue'

describe('Plan.vue v-permission', () => {
  it('should hide edit buttons for duty officer role', () => {
    const wrapper = mount(Plan.vue, {
      global: {
        directives: { permission: PermissionDirective },
        mocks: {
          $route: { meta: { roles: ['duty_officer'] } }
        }
      }
    })
    
    // 编辑按钮应隐藏（display:none）
    expect(wrapper.find('[data-testid="edit-btn"]').element).toHaveStyle('display: none')
  })
  
  it('should show create button for chief role', () => {
    const wrapper = mount(Plan.vue, {
      global: {
        directives: { permission: PermissionDirective },
        mocks: {
          $route: { meta: { roles: ['chief'] } }
        }
      }
    })
    
    // 创建按钮应可见
    expect(wrapper.find('[data-testid="create-btn"]').exists()).toBe(true)
  })
})
```
