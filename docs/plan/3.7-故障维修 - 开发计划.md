## 一、需求范围

| 编号 | 功能点 | 需求描述 | 优先级 | 开发计划边界 |
|------|--------|----------|--------|--------------|

## 二、数据模型设计

### 2.1 数据库表结构（基于 PRD line 521-535）

```sql
CREATE TABLE repair_orders (
    id BIGSERIAL PRIMARY KEY,
    order_no VARCHAR(50) UNIQUE NOT NULL,  -- RO-20260910-001
    device_id BIGINT REFERENCES devices(id),
    alarm_id BIGINT REFERENCES alarms(id),  -- 可选：告警转工单时关联
    inspection_record_id BIGINT REFERENCES inspection_records(id),  -- 新增：巡检转工单时关联
    fault_desc TEXT NOT NULL,
    status VARCHAR(20) DEFAULT 'pending',  -- pending / assigned / repairing / pending_accept / completed / returned
    reporter_id BIGINT REFERENCES users(id),  -- 报修人（巡检责任人/值班员）
    repairer_id BIGINT REFERENCES users(id),  -- 维修人员
    acceptor_id BIGINT REFERENCES users(id),  -- 新增：验收人（消防主管）
    assigned_at TIMESTAMPTZ,
    completed_at TIMESTAMPTZ,
    accepted_at TIMESTAMPTZ,  -- 新增：验收时间
    repair_result TEXT,  -- 维修结果描述（含配件明细）
    return_reason TEXT,  -- 新增：退回原因
    created_by BIGINT REFERENCES users(id),  -- 创建人（审计用途）
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 索引优化
CREATE INDEX idx_repair_orders_device ON repair_orders(device_id);
CREATE INDEX idx_repair_orders_status ON repair_orders(status);
CREATE INDEX idx_repair_orders_repairer ON repair_orders(repairer_id);
CREATE INDEX idx_repair_orders_created ON repair_orders(created_at);
```

### 2.2 状态枚举定义

```python
# 工单状态枚举（6 种）
REPAIR_ORDER_STATUS = {

    'assigned': '已派单',          # 已分配维修人员
    'repairing': '维修中',         # 维修进行中
    'pending_accept': '待验收',    # 维修完成，等待验收

    'returned': '已退回'           # 验收不通过，退回重新维修

# 状态流转图

"""
```

## 四、核心业务逻辑

### 4.1 巡检异常自动转工单

```python
# 触发条件：巡检记录 result='abnormal'
async def auto_create_repair_order_from_inspection(inspection_record_id: int):
    """
    巡检异常自动创建维修工单
    """
    # 1. 获取巡检记录
    record = await get_inspection_record(inspection_record_id)
    if record.result != 'abnormal':
        return
    
    # 2. 检查是否已存在关联工单（避免重复创建）
    existing = await get_repair_order_by_inspection_record(inspection_record_id)
    if existing:
        return
    
    # 3. 创建工单
    order = RepairOrder(
        order_no=generate_order_no(),  # RO-YYYYMMDD-XXX
        device_id=record.device_id,
        inspection_record_id=record.id,
        fault_desc=record.abnormal_desc or "巡检发现异常",
        status='pending',
        reporter_id=record.inspected_by,  # 巡检责任人作为报修人
        created_by=record.created_by
    )
    db.add(order)
    await db.commit()
    
    # 4. 发送通知给消防主管
    await notify_chief_new_repair_order(order)
```

### 4.2 工单状态流转

```python
# 状态流转规则
STATUS_TRANSITIONS = {
    'pending': ['assigned'],
    'assigned': ['repairing'],
    'repairing': ['pending_accept'],
    'pending_accept': ['completed', 'returned'],
    'returned': ['repairing'],  # 退回后重新维修
    'completed': []  # 终态
}

# 派单
async def assign_repair_order(order_id: int, repairer_id: int, current_user: User):
    order = await get_repair_order(order_id)
    if order.status != 'pending':
        raise ValidationError("只能派单待处理工单")
    
    order.repairer_id = repairer_id
    order.status = 'assigned'
    order.assigned_at = datetime.now()
    await db.commit()
    
    # 通知维修人员
    await notify_repairer_assigned(order, repairer_id)

# 完成维修
async def complete_repair_order(order_id: int, repair_result: str, current_user: User):
    order = await get_repair_order(order_id)
    if order.status != 'repairing':
        raise ValidationError("只能完成维修中的工单")
    
    order.repair_result = repair_result
    order.status = 'pending_accept'
    order.completed_at = datetime.now()
    await db.commit()
    
    # 通知消防主管验收
    await notify_chief_pending_accept(order)

# 验收通过
async def accept_repair_order(order_id: int, current_user: User):
    order = await get_repair_order(order_id)
    if order.status != 'pending_accept':
        raise ValidationError("只能验收待验收工单")
    
    # 验证权限：仅消防主管可验收
    if 'chief' not in [r.role_code for r in current_user.roles]:
        raise PermissionDenied("仅消防主管可执行验收")
    
    order.acceptor_id = current_user.id
    order.status = 'completed'
    order.accepted_at = datetime.now()
    await db.commit()
    
    # 恢复设备状态
    await update_device_status(order.device_id, 'normal')

# 验收退回
async def return_repair_order(order_id: int, return_reason: str, current_user: User):
    order = await get_repair_order(order_id)
    if order.status != 'pending_accept':
        raise ValidationError("只能退回待验收工单")
    
    # 验证权限
    if 'chief' not in [r.role_code for r in current_user.roles]:
        raise PermissionDenied("仅消防主管可执行验收")
    
    order.return_reason = return_reason
    order.status = 'returned'
    order.acceptor_id = current_user.id
    await db.commit()
    
    # 通知维修人员重新维修
    await notify_repairer_returned(order, return_reason)
```

### 4.3 统计接口实现

```python
# 平均维修时长
async def get_avg_repair_duration(date_range: str = '30d'):
    """
    计算指定时间范围内的平均维修时长（小时）
    """
    start_date = parse_date_range(date_range)
    
    query = select(
        func.avg(
            func.extract('epoch', repair_orders.completed_at - repair_orders.assigned_at) / 3600
        )
    ).where(
        repair_orders.status == 'completed',
        repair_orders.completed_at >= start_date
    )
    
    result = await db.execute(query)
    return {'avg_hours': result.scalar() or 0}

# 故障类型分布
async def get_fault_distribution(date_range: str = '30d'):
    """
    按故障描述关键词分组统计
    """
    start_date = parse_date_range(date_range)
    
    # 简化实现：按设备类型分组
    query = select(
        device_types.type_name,
        func.count(repair_orders.id)
    ).join(
        devices, repair_orders.device_id == devices.id
    ).join(
        device_types, devices.type_id == device_types.id
    ).where(
        repair_orders.created_at >= start_date
    ).group_by(device_types.type_name)
    
    result = await db.execute(query)
    return [{'type': row[0], 'count': row[1]} for row in result]

# 维修人员工作量
async def get_repairer_workload(date_range: str = '30d'):
    """
    统计各维修人员的工单数量
    """
    start_date = parse_date_range(date_range)
    
    query = select(
        users.real_name,
        func.count(repair_orders.id),
        func.sum(
            case((repair_orders.status == 'completed', 1), else_=0)
        )
    ).join(
        repair_orders, users.id == repair_orders.repairer_id
    ).where(
        repair_orders.assigned_at >= start_date
    ).group_by(users.id, users.real_name)
    
    result = await db.execute(query)
    return [
        {'name': row[0], 'total': row[1], 'completed': row[2]}
        for row in result
    ]

# 故障设备 TOP10
async def get_top10_fault_devices(date_range: str = '30d'):
    """
    统计故障次数最多的前 10 个设备
    """
    start_date = parse_date_range(date_range)
    
    query = select(
        devices.device_code,
        devices.device_name,
        device_types.type_name,
        func.count(repair_orders.id).label('fault_count')
    ).join(
        repair_orders, devices.id == repair_orders.device_id
    ).join(
        device_types, devices.type_id == device_types.id
    ).where(
        repair_orders.created_at >= start_date
    ).group_by(
        devices.id, devices.device_code, devices.device_name, device_types.type_name
    ).order_by(
        desc('fault_count')
    ).limit(10)
    
    result = await db.execute(query)
    return [
        {
            'device_code': row[0],
            'device_name': row[1],
            'type_name': row[2],
            'fault_count': row[3]
        }
        for row in result
    ]
```

## 八、验收标准

### 8.1 功能验收

- [ ] 工单可从 `pending` 流转到 `completed`，支持 `returned` 退回
- [ ] 巡检异常记录自动创建工单，默认分配给巡检责任人
- [ ] 维修完成后设备状态自动恢复为 `normal`
- [ ] 仅消防主管可执行验收操作
- [ ] 验收不通过时，工单退回至 `repairing` 状态
- [ ] 4 项统计指标数据准确

### 8.2 性能验收

- [ ] 工单列表页加载时间 ≤ 2 秒（1000 条数据）
- [ ] 统计接口响应时间 ≤ 3 秒（30 天数据）
- [ ] TOP10 统计支持 10,000+ 工单数据
